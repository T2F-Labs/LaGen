% ========== FONT CONFIGURATION ==========
% Professional font setup using fontspec for XeLaTeX
% Body text: TeX Gyre Pagella (Palatino-like serif)
% Code: DejaVu Sans Mono (monospace)
% Headings: TeX Gyre Heros (Helvetica-like sans-serif)
% Math: Handled by mathematics module when enabled

% Set main document font (body text)
\setmainfont{TeX Gyre Pagella}[
    Ligatures=TeX,
    Numbers=OldStyle
]

% Set sans-serif font (headings)
\setsansfont{TeX Gyre Heros}[
    Ligatures=TeX,
    Scale=MatchLowercase
]

% Set monospace font (code)
\setmonofont{DejaVu Sans Mono}[
    Scale=MatchLowercase
]

\typeout{^^J INFO: Font configuration loaded - TeX Gyre Pagella, TeX Gyre Heros, DejaVu Sans Mono^^J}

% ========== CHAPTER-BASED COUNTER CONFIGURATION ==========
% Configure figures and tables to reset and number by chapter
% This must be done AFTER all packages are loaded
\counterwithin{figure}{chapter}
\counterwithin{table}{chapter}
\renewcommand{\thefigure}{\thechapter.\arabic{figure}}
\renewcommand{\thetable}{\thechapter.\arabic{table}}

% Add "Figure" and "Table" prefix in List of Figures and List of Tables
\renewcommand{\cftfigpresnum}{Figure }
\renewcommand{\cfttabpresnum}{Table }
\setlength{\cftfignumwidth}{5em}  % Adjust width to accommodate "Figure X.Y"
\setlength{\cfttabnumwidth}{5em}  % Adjust width to accommodate "Table X.Y"

\typeout{^^J INFO: Configured chapter-based numbering for figures and tables^^J}

% ========== TABLE OF CONTENTS DEPTH ==========
% Control what appears in the table of contents
% 0 = chapters only
% 1 = chapters and sections
% 2 = chapters, sections, and subsections (default)
% 3 = chapters, sections, subsections, and subsubsections
\setcounter{tocdepth}{2}  % Show chapters, sections, and subsections
\setcounter{secnumdepth}{3}  % But number up to subsubsections in the text
\typeout{^^J INFO: TOC depth set to 2 (chapters, sections, and subsections)^^J}
