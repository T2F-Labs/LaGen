\section*{Extension Manifest Schema}
\addcontentsline{toc}{section}{Extension Manifest Schema}

The Symphony.toml manifest file defines extension metadata, dependencies, permissions, and configuration schema. This specification ensures consistent extension packaging and enables Symphony's security and orchestration systems to properly manage extensions.

\subsection*{Manifest File Structure}

The complete Symphony.toml schema with all available sections:

\begin{lstlisting}[language=toml,caption={Complete Extension Manifest Schema}]
# Extension metadata
[extension]
name = "my-extension"
version = "1.0.0"
description = "A Symphony extension"
author = "Developer Name <developer@example.com>"
license = "MIT"
homepage = "https://github.com/developer/my-extension"
repository = "https://github.com/developer/my-extension"
documentation = "https://docs.example.com/my-extension"
keywords = ["ai", "development", "productivity"]
categories = ["instruments", "productivity", "ai-models"]

# Minimum Symphony version required
symphony_version = ">=1.0.0"

# Extension type and execution model
type = "instrument"  # Options: "instrument", "operator", "motif"
execution_model = "out_of_process"  # Options: "in_process", "out_of_process"

# Entry points
[extension.entry_points]
main = "src/main.rs"
ui = "src/ui/index.tsx"
config = "src/config.rs"
\end{lstlisting}

\subsection*{Configuration Schema}

Define extension configuration options and UI generation:

\begin{lstlisting}[language=toml,caption={Configuration Schema}]
[config]
# Configuration schema for auto-generated UI
[config.schema]
[config.schema.api_key]
type = "string"
title = "API Key"
description = "Your API key for the service"
required = true
secret = true
default = ""

[config.schema.model_name]
type = "string"
title = "Model Name"
description = "Name of the AI model to use"
required = true
default = "gpt-4"
enum = ["gpt-4", "gpt-3.5-turbo", "claude-3"]

[config.schema.temperature]
type = "number"
title = "Temperature"
description = "Sampling temperature for model responses"
required = false
default = 0.7
minimum = 0.0
maximum = 2.0
step = 0.1

[config.schema.max_tokens]
type = "integer"
title = "Max Tokens"
description = "Maximum number of tokens in response"
required = false
default = 1000
minimum = 1
maximum = 4000

[config.schema.enable_caching]
type = "boolean"
title = "Enable Caching"
description = "Cache responses to improve performance"
required = false
default = true

[config.schema.custom_prompts]
type = "array"
title = "Custom Prompts"
description = "List of custom prompt templates"
required = false
items = { type = "string" }
default = []

[config.schema.advanced_settings]
type = "object"
title = "Advanced Settings"
description = "Advanced configuration options"
required = false
properties = {
    timeout = { type = "integer", default = 30000 },
    retry_count = { type = "integer", default = 3 },
    debug_mode = { type = "boolean", default = false }
}

# Configuration validation
[config.validation]
# Custom validation rules
rules = [
    "api_key.length >= 10",
    "temperature >= 0.0 && temperature <= 2.0",
    "max_tokens > 0"
]

# Configuration groups for UI organization
[config.groups]
[config.groups.authentication]
title = "Authentication"
description = "API keys and authentication settings"
fields = ["api_key"]

[config.groups.model]
title = "Model Settings"
description = "AI model configuration"
fields = ["model_name", "temperature", "max_tokens"]

[config.groups.performance]
title = "Performance"
description = "Performance and caching options"
fields = ["enable_caching", "advanced_settings"]
\end{lstlisting}

\subsection*{Build and Packaging}

Build configuration and packaging options:

\begin{lstlisting}[language=toml,caption={Build Configuration}]
[build]
# Build system configuration
system = "cargo"  # Options: "cargo", "npm", "webpack", "custom"
target = "wasm32-unknown-unknown"
profile = "release"

# Build flags and options
[build.flags]
rust = [
    "--features", "symphony-integration",
    "--target-dir", "target"
]

javascript = [
    "--mode", "production",
    "--optimize-minimize"
]

# Build environment
[build.env]
SYMPHONY_API_VERSION = "1.0"
EXTENSION_VERSION = "${extension.version}"

# Pre/post build scripts
[build.scripts]
pre_build = ["npm install", "cargo check"]
post_build = ["npm run test", "cargo test"]

# Asset processing
[build.assets]
include = [
    "assets/**/*",
    "README.md",
    "LICENSE"
]
exclude = [
    "src/**/*.test.*",
    "**/.DS_Store",
    "**/node_modules"
]

# Packaging options
[package]
# Package format
format = "syx"  # Symphony extension format
compression = "gzip"
include_source = false

# Code signing
[package.signing]
required = true
key_file = "private.key"
cert_file = "certificate.crt"

# Distribution
[package.distribution]
channels = ["stable", "beta"]
auto_update = true
update_check_url = "https://api.example.com/updates"
\end{lstlisting}

This manifest schema enables Symphony's extension system to properly manage, secure, and orchestrate extensions while providing rich metadata for discovery and configuration. The schema supports both simple extensions and complex AI-powered tools within Symphony's Dual Ensemble Architecture.
