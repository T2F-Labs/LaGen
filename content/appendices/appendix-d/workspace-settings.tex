\section*{Workspace Settings}
\addcontentsline{toc}{section}{Workspace Settings}

Workspace settings provide project-specific configuration that overrides global user settings. These settings are stored within the workspace directory and can be version-controlled alongside project code, enabling reproducible development environments for research validation.

\subsection*{Workspace Configuration File}

Workspace settings are stored in \texttt{.symphony/workspace.toml} within the project root:

\begin{lstlisting}[language=toml,caption={Workspace Configuration Structure}]
[workspace]
name = "My Symphony Project"
version = "1.0.0"
description = "AI-first development project"
created = "2024-01-15T10:30:00Z"
last_modified = "2024-01-20T14:45:00Z"

# Workspace-specific settings
root_path = "."
exclude_patterns = [
    "**/node_modules/**",
    "**/.git/**",
    "**/target/**",
    "**/__pycache__/**"
]

# Project metadata
[workspace.metadata]
language = "rust"
framework = "symphony"
team = ["developer1", "developer2"]
tags = ["ai", "development", "research"]
\end{lstlisting}

Override global editor settings for project consistency:

\begin{lstlisting}[language=toml,caption={Project Editor Configuration}]

\end{lstlisting}

\subsection*{Extension Configuration}

Workspace-specific extension settings and requirements:

\begin{lstlisting}[language=toml,caption={Workspace Extension Configuration}]
[extensions]

required = [
    "rust-analyzer",
    "symphony-ai-assistant",
    "git-integration"
]

# Recommended extensions
recommended = [
    "markdown-preview",
    "toml-syntax",
    "performance-profiler"
]

# Extension-specific settings
[extensions.rust-analyzer]
check_on_save = true
cargo_features = ["ai", "performance"]
proc_macro_srv = true

[extensions.symphony-ai-assistant]
auto_suggestions = true
context_awareness = true
learning_mode = true

[extensions.git-integration]
auto_fetch = true
show_inline_blame = false
commit_template = "feat: ${summary}\n\n${description}"

# Workspace-specific extension permissions
[extensions.permissions]
"rust-analyzer" = [
    "read_workspace",
    "write_artifacts",
    "execute_commands"
]
"symphony-ai-assistant" = [
    "read_workspace",
    "write_artifacts",
    "network_access",
    "ai_models"
]
\end{lstlisting}

\subsection*{Build and Task Configuration}

Project build settings and task definitions:

\begin{lstlisting}[language=toml,caption={Build and Task Configuration}]
[build]
# Build system configuration
system = "cargo"  # Options: "cargo", "npm", "make", "custom"
target = "x86_64-unknown-linux-gnu"
profile = "dev"  # Options: "dev", "release", "test"

# Build flags and options
[build.flags]

    "--features", "ai,performance",

]

# Environment variables for builds
[build.env]
RUST_LOG = "debug"

\subsection*{Debug and Launch Configuration}

Debugging and launch configurations for development:

\begin{lstlisting}[language=toml,caption={Debug Configuration}]
[debug]
# Default debug configuration
default_configuration = "rust-debug"

# Debug configurations
[debug.configurations]
[debug.configurations.rust-debug]
type = "rust"
request = "launch"
program = "${workspaceFolder}/target/debug/symphony"
args = ["--debug", "--workspace", "${workspaceFolder}"]
env = { RUST_LOG = "debug" }
console = "integratedTerminal"

[debug.configurations.rust-test]
type = "rust"
request = "launch"
program = "cargo"
args = ["test", "--", "--nocapture"]
env = { RUST_LOG = "trace" }

[debug.configurations.python-ai]
type = "python"
request = "launch"
program = "${workspaceFolder}/src/ai/conductor.py"
args = ["--debug"]
env = { PYTHONPATH = "${workspaceFolder}/src" }

# Launch configurations
[launch]
[launch.configurations]
[launch.configurations.development]
name = "Development Server"
command = "cargo run"
args = ["--features", "dev"]
env = { SYMPHONY_ENV = "development" }

[launch.configurations.production]
name = "Production Build"
command = "cargo run --release"
env = { SYMPHONY_ENV = "production" }
\end{lstlisting}

\subsection*{Testing Configuration}

Test framework and coverage settings:

\begin{lstlisting}[language=toml,caption={Testing Configuration}]
[testing]
# Test framework settings
framework = "cargo-test"
parallel = true
timeout = 30000  # milliseconds

# Test discovery
test_patterns = [
    "tests/**/*.rs",
    "src/**/*_test.rs",
    "src/**/test_*.rs"
]

exclude_patterns = [
    "target/**",
    "examples/**"
]

# Coverage settings
[testing.coverage]
enabled = true
tool = "tarpaulin"  # Options: "tarpaulin", "grcov", "kcov"
output_format = ["html", "lcov"]
output_dir = "coverage"
minimum_coverage = 80.0

# Test categories
[testing.categories]
unit = { pattern = "src/**/*_test.rs" }
integration = { pattern = "tests/**/*.rs" }
benchmark = { pattern = "benches/**/*.rs" }
ai = { pattern = "tests/ai/**/*.rs" }

# Test environment
[testing.env]
RUST_LOG = "debug"
TEST_MODE = "true"
AI_MODEL_PATH = "./test_models"
\end{STlisting}

\subsection*{Git and Version Control}

Version control settings and hooks:

\begin{lstlisting}[language=toml,caption={Git Configuration}]
[git]
# Git integration settings
auto_fetch = true
fetch_interval = 300  # seconds
show_status = true
inline_blame = false

# Commit settings
[git.commit]
template = """
${type}: ${summary}

${description}

Co-authored-by: ${co_authors}
"""

# Commit types
types = [
    "feat",     # New feature
    "fix",      # Bug fix
    "docs",     # Documentation
    "style",    # Code style
    "refactor", # Code refactoring
    "test",     # Tests
    "chore"     # Maintenance
]

# Pre-commit hooks
[git.hooks]
pre_commit = [
    "cargo fmt --check",
    "cargo clippy -- -D warnings",
    "cargo test"
]

pre_push = [
    "cargo test --release",
    "cargo doc"
]

# Branch protection
[git.branches]
main = { protected = true, require_pr = true }
develop = { protected = false, auto_merge = true }

# Ignore patterns (in addition to .gitignore)
[git.ignore]
patterns = [
    "*.tmp",
    ".symphony/cache/",
    "coverage/",
    "docs/build/"
]
\end{lstlisting}

Workspace settings enable project-specific customization while maintaining consistency across team members and development environments. These settings support Symphony's research objectives by providing reproducible, version-controlled configuration that can be validated and shared alongside research artifacts.
