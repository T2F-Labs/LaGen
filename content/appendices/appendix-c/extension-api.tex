\section*{Extension API}
\addcontentsline{toc}{section}{Extension API}

Symphony's Extension API provides the foundational interfaces for developing extensions within the Dual Ensemble Architecture. The API design emphasizes type safety, performance, and clear separation between The Pit (infrastructure extensions) and The Grand Stage (user-facing extensions).

\subsection*{Core Extension Trait}

The fundamental interface that all Symphony extensions must implement:

\begin{lstlisting}[language=Rust,caption={Core Extension Trait Definition}]
pub trait Extension {
    /// Extension metadata and capabilities
    fn metadata(&self) -> ExtensionMetadata;
    
    /// Initialize extension with provided context
    fn initialize(&mut self, context: &ExtensionContext) -> Result<(), ExtensionError>;
    
    /// Handle incoming requests from the Conductor
    fn handle_request(&mut self, request: Request) -> Result<Response, ExtensionError>;
    
    /// Cleanup resources before shutdown
    fn shutdown(&mut self) -> Result<(), ExtensionError>;
    
    /// Health check for monitoring
    fn health_check(&self) -> HealthStatus;
}
\end{lstlisting}

pub struct ExtensionContext {

    pub config: ExtensionConfig,
    pub logger: Logger,
    pub event_bus: EventBus,

impl ExtensionContext {
    /// Access to Symphony's artifact storage

\subsection*{Event System}

Publish-subscribe event system for extension communication:

\begin{lstlisting}[language=Rust,caption={Event System Interface}]
pub trait EventBus {
    /// Subscribe to events matching pattern
    fn subscribe<T>(&self, pattern: &str, handler: EventHandler<T>) -> SubscriptionId
    where
        T: DeserializeOwned + Send + 'static;
    
    /// Publish event to subscribers
    fn publish<T>(&self, event_type: &str, payload: &T) -> Result<(), EventError>
    where
        T: Serialize;
    
    /// Unsubscribe from events
    fn unsubscribe(&self, subscription_id: SubscriptionId) -> Result<(), EventError>;
}

pub type EventHandler<T> = Box<dyn Fn(&T) -> Result<(), EventError> + Send + Sync>;
\end{lstlisting}

\subsection*{Error Handling}

Comprehensive error types for extension development:

\begin{lstlisting}[language=Rust,caption={Extension Error Types}]
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ExtensionError {
    InitializationFailed(String),
    InvalidRequest(String),
    PermissionDenied(String),
    ResourceUnavailable(String),
    InternalError(String),
    TimeoutError(String),
    ConfigurationError(String),
}

impl std::error::Error for ExtensionError {}
impl std::fmt::Display for ExtensionError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            ExtensionError::InitializationFailed(msg) => write!(f, "Initialization failed: {}", msg),
            ExtensionError::InvalidRequest(msg) => write!(f, "Invalid request: {}", msg),
            ExtensionError::PermissionDenied(msg) => write!(f, "Permission denied: {}", msg),
            ExtensionError::ResourceUnavailable(msg) => write!(f, "Resource unavailable: {}", msg),
            ExtensionError::InternalError(msg) => write!(f, "Internal error: {}", msg),
            ExtensionError::TimeoutError(msg) => write!(f, "Timeout error: {}", msg),
            ExtensionError::ConfigurationError(msg) => write!(f, "Configuration error: {}", msg),
        }
    }
}
\end{lstlisting}

\subsection*{Usage Examples}

\subsubsection*{Basic Extension Implementation}

\begin{lstlisting}[language=Rust,caption={Simple Extension Example}]
use symphony_extension_api::*;

pub struct HelloWorldExtension {
    context: Option<ExtensionContext>,
}

impl Extension for HelloWorldExtension {
    fn metadata(&self) -> ExtensionMetadata {
        ExtensionMetadata {
            id: "hello-world".into(),
            name: "Hello World Extension".into(),
            version: "1.0.0".parse().unwrap(),
            description: "A simple example extension".into(),
            author: "Symphony Team".into(),
            license: "MIT".into(),
            capabilities: vec![],
            dependencies: vec![],
            permissions: vec![Permission::ReadWorkspace],
            execution_model: ExecutionModel::OutOfProcess,
        }
    }
    
    fn initialize(&mut self, context: &ExtensionContext) -> Result<(), ExtensionError> {
        context.log(LogLevel::Info, "Hello World extension initialized");
        self.context = Some(context.clone());
        Ok(())
    }
    
    fn handle_request(&mut self, request: Request) -> Result<Response, ExtensionError> {
        match request.method.as_str() {
            "greet" => {
                let response_data = json!({"message": "Hello from Symphony!"});
                Ok(Response {
                    id: request.id,
                    result: Some(response_data),
                    error: None,
                    metadata: ResponseMetadata::default(),
                })
            }
            _ => Err(ExtensionError::InvalidRequest(
                format!("Unknown method: {}", request.method)
            ))
        }
    }
    
    fn shutdown(&mut self) -> Result<(), ExtensionError> {
        if let Some(context) = &self.context {
            context.log(LogLevel::Info, "Hello World extension shutting down");
        }
        Ok(())
    }
    
    fn health_check(&self) -> HealthStatus {
        HealthStatus::Healthy
    }
}
\end{lstlisting}
